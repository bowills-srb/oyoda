    # Operator credentials — loaded from env vars for security
    # Set OPERATOR_1_EMAIL, OPERATOR_1_PASSWORD, OPERATOR_1_ID in Railway
    DEMO_OPERATORS = {}

    # Load operators from environment (supports up to 5 operators)
    for i in range(1, 6):
        op_email = os.getenv(f"OPERATOR_{i}_EMAIL", "").strip().lower()
        op_password = os.getenv(f"OPERATOR_{i}_PASSWORD", "")
        op_id = os.getenv(f"OPERATOR_{i}_ID", f"op_{i:03d}")
        op_name = os.getenv(f"OPERATOR_{i}_NAME", "Operator")
        op_company = os.getenv(f"OPERATOR_{i}_COMPANY", "Properties")
        if op_email and op_password:
            DEMO_OPERATORS[op_email] = {
                "id": op_id,
                "password": op_password,
                "name": op_name,
                "company": op_company,
                "role": "owner",
                "properties": 50,
                "pms": "Escapia"
            }

    # Fallback demo account (only active if no env operators configured)
    if not DEMO_OPERATORS:
